"""Coordinator checks frozen review bytes and independently reads own DB copies."""
import ast
import hashlib
import io
import json
import sqlite3
import subprocess
from collections import Counter
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path('D:/1800')
HERE = Path(__file__).resolve().parent
REVIEW = ROOT / 'docs/evidence/A03/review/P2-R12-R1-failure'
TARGET = '6affc61d2ab0b1f0d96bb2497c5ae969ebaf1df2'
REPORT = 'docs/tasks/A03-P2-R12-R1-failure-review-report.md'


def sha(data):
    return hashlib.sha256(data).hexdigest()


def readj(path):
    return json.loads(path.read_bytes())


def git(*args, data=None):
    return subprocess.run(['git', '-c', 'core.longpaths=true', *args], cwd=ROOT,
                          input=data, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                          check=True, timeout=60).stdout


def fetch(specs):
    stream = io.BytesIO(git('cat-file', '--batch', data=('\n'.join(specs) + '\n').encode('utf-8')))
    result = {}
    for spec in specs:
        header = stream.readline().split()
        assert len(header) == 3 and header[1] == b'blob', spec
        result[spec] = stream.read(int(header[2]))
        assert stream.read(1) == b'\n'
    assert not stream.read()
    return result


def main():
    expected = {
        ROOT / REPORT: '145412bb9b81958d4615ff47d0943956552da3e73649cd5afc0c8731934ba227',
        REVIEW / 'sha256-manifest.json': '562b17a188c093dd74c9fab84b8b0deece7c64a0a547f799964534d803e2d015',
        REVIEW / 'process-release.json': '02d6781ff07705159de3f693e7658a8594b5f1285a7895fc44b7693b1d7c0ebb',
    }
    for path, digest in expected.items():
        assert sha(path.read_bytes()) == digest
    manifest = readj(REVIEW / 'sha256-manifest.json')
    assert manifest['target'] == TARGET and manifest['functional_acceptance'] is False
    files = {x['path']: x for x in manifest['files']}
    for path, entry in files.items():
        data = (ROOT / path).read_bytes()
        assert len(data) == entry['bytes'] and sha(data) == entry['sha256'], path
    actual = {p.relative_to(ROOT).as_posix() for p in REVIEW.rglob('*') if p.is_file()}
    assert actual | {REPORT} == set(files) | {manifest['excluded_only']}
    exports = readj(REVIEW / 'input-index.json')['exports']
    raw = fetch(sorted({x['commit'] + ':' + x['path'] for x in exports}))
    for entry in exports:
        data = raw[entry['commit'] + ':' + entry['path']]
        assert data == (REVIEW / entry['file']).read_bytes()
        assert len(data) == entry['bytes'] and sha(data) == entry['sha256']
    release = readj(REVIEW / 'process-release.json')
    assert release['current_registered_children'] == len(release['records']) == 9
    for entry in release['records']:
        folder = REVIEW / entry['record']
        start, end = readj(folder / 'started.json'), readj(folder / 'completed.json')
        assert start['pid'] == end['pid'] == entry['pid']
        assert end['exit_code'] == entry['exit_code'] == 0
        assert end['alive'] is entry['alive'] is False
        assert end['timed_out'] is entry['timed_out'] is False
        for kind in ['stdout', 'stderr']:
            streams = [p for p in folder.iterdir() if p.stem == kind]
            assert len(streams) == 1
            assert sha(streams[0].read_bytes()) == end[kind + '_sha256'] == entry[kind + '_sha256']

    # Independent queries use only newly written coordinator copies, never the review/author originals.
    E = 'docs/evidence/A03/execution/P2-R12-R1-unknown-projection/validation/author-01/cases/'
    paths = sorted(x['path'] for x in exports if x['commit'] == TARGET
                   and x['path'].startswith(E) and '/captures/' in x['path'] and x['path'].endswith('.sqlite3'))
    assert len(paths) == 12
    tables = {}
    db_results = []
    for path in paths:
        data = raw[TARGET + ':' + path]
        scene, _, stage, filename = path[len(E):].split('/')
        own = HERE / 'databases' / scene / stage / filename
        own.parent.mkdir(parents=True, exist_ok=True)
        assert not own.exists()
        own.write_bytes(data)
        conn = sqlite3.connect(own.as_uri() + '?mode=ro&immutable=1', uri=True)
        conn.row_factory = sqlite3.Row
        try:
            conn.execute('PRAGMA query_only=ON')
            assert conn.execute('PRAGMA quick_check').fetchone()[0] == 'ok'
            names = [r[0] for r in conn.execute("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name")]
            all_rows = {name: [dict(row) for row in conn.execute('SELECT * FROM "' + name.replace('"', '""') + '"')]
                        for name in names}
        finally:
            conn.close()
        assert own.read_bytes() == data
        tables[(scene, stage, filename)] = all_rows
        db_results.append({'source': path, 'copy': own.relative_to(ROOT).as_posix(), 'sha256': sha(data),
                           'counts': {name: len(rows) for name, rows in all_rows.items()},
                           'selected_rows': {name: all_rows[name] for name in ['actions', 'intents', 'permits', 'plans', 'books'] if name in all_rows}})
    u = tables[('U1', 'first-failure', 'runtime.sqlite3')]
    ur = tables[('U1', 'after-release', 'runtime.sqlite3')]
    assert [r['state'] for r in u['actions']] == ['UNKNOWN']
    assert [r['state'] for r in u['intents']] == ['PENDING'] and u['intents'][0]['receipt'] is None
    assert [r['state'] for r in u['permits']] == ['CONSUMED']
    assert [r['status'] for r in u['plans']] == ['INTERRUPTED']
    assert not u['native_report_reads'] and not u['calls'] and not u['attempts'] and not u['resolutions']
    kinds = Counter(json.loads(r['ref'])['kind'] for r in u['artifacts'])
    assert kinds['ExecutionReceipt'] == kinds['ExecutionProgress'] == kinds['BatchReport'] == 0
    assert ur['books'][0]['owner'] is None and ur['books'][0]['pid'] is None
    assert ur['books'][0]['phase'] == 'RELEASED' and ur['books'][0]['lease'] == 0
    for name in ['actions', 'intents', 'permits', 'plans']:
        assert u[name] == ur[name]
    ut = tables[('U1', 'first-failure', 'target.sqlite3')]
    assert ut == tables[('U1', 'after-release', 'target.sqlite3')]
    assert len(ut['effects']) == len(ut['questions']) == len(ut['frames']) == 1
    l = tables[('L1', 'unknown-established', 'runtime.sqlite3')]
    lk = Counter(json.loads(r['ref'])['kind'] for r in l['artifacts'])
    assert lk['ExecutionReceipt'] == lk['ExecutionProgress'] == 1 and lk['BatchReport'] == 0

    source = 'src/annotation/runtime/storage/__init__.py'
    data = raw[TARGET + ':' + source]
    module = ast.parse(data)
    cls = next(n for n in module.body if isinstance(n, ast.ClassDef) and n.name == 'RuntimeStore')
    method = next(n for n in cls.body if isinstance(n, ast.FunctionDef) and n.name == '_project_plan_locked')
    append_lines = [n.lineno for n in ast.walk(method) if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute)
                    and n.func.attr == '_append' and len(n.args) >= 3 and isinstance(n.args[2], ast.Name) and n.args[2].id == 'receipt']
    policy_lines = [n.lineno for n in ast.walk(method) if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute)
                    and n.func.attr == '_project_unknown_diagnostic_post']
    assert append_lines == [792] and policy_lines == [794]
    answer = {'at': datetime.now(timezone.utc).isoformat(), 'target': TARGET,
              'decision': 'FAILURE_AND_REPAIR_BOUNDARY_CORROBORATED', 'functional_acceptance': False,
              'manifest_members_verified': len(files), 'raw_exports_verified': len(exports),
              'review_processes_exited': 9, 'artifacts': {p.relative_to(ROOT).as_posix(): v for p, v in expected.items()},
              'own_immutable_stage_DB_queries': db_results, 'U1_formal_kinds': dict(kinds),
              'L1_formal_kinds': dict(lk), 'receipt_append_line': append_lines[0], 'later_report_policy_line': policy_lines[0],
              'product_calls': 0, 'functional_runs': 0, 'original_DB_connections': 0,
              'limits': ['Only frozen synthetic evidence, no platform or functional acceptance.',
                         'Sequential per-database snapshots, not cross-database atomic.',
                         'Whole A03 and I01 remain unaccepted.']}
    out = HERE / 'r12-r1-failure-corroboration.json'
    out.write_bytes((json.dumps(answer, ensure_ascii=False, indent=2) + '\n').encode('utf-8'))
    print(json.dumps({'path': str(out), 'sha256': sha(out.read_bytes()),
                      'manifest': len(files), 'exports': len(exports), 'DBs': len(paths)}))


if __name__ == '__main__':
    main()
